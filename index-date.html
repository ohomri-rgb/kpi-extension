(function() { const VERSION = "v1.1.2"; let hasUpdatedDate = false; const checkTableauLoaded = setInterval(() => { if (typeof window.tableau !== 'undefined' && window.tableau.extensions) { clearInterval(checkTableauLoaded); initializeDateUpdater(); } }, 50); function initializeDateUpdater() { window.tableau.extensions.initializeAsync().then(() => { if (hasUpdatedDate) return; const dashboard = window.tableau.extensions.dashboardContent.dashboard; const today = new Date(); today.setHours(0, 0, 0, 0); let debugLog = `גרסה: ${VERSION}
מתחיל סריקת גיליונות...
`; addStatusToUI(debugLog); // רצים על כל הגיליונות בדשבורד const promises = dashboard.worksheets.map(worksheet => { return worksheet.getFiltersAsync().then(filters => { debugLog += `🔍 גיליון: "${worksheet.name}" (נמצאו ${filters.length} פילטרים)
`; filters.forEach(f => { debugLog += ` • פילטר: "${f.fieldName}" | סוג: ${f.filterType} | דאטה: ${f.dataType}
`; }); addStatusToUI(debugLog); // מחפשים כל פילטר שקשור לתאריך const dateFilter = filters.find(filter => filter.dataType === window.tableau.DataType.Date || filter.dataType === window.tableau.DataType.DateTime || filter.fieldName.toLowerCase().includes('date') || filter.fieldName.includes('תאריך') ); if (dateFilter) { debugLog += `⚡ מנסה לעדכן את "${dateFilter.fieldName}" בגיליון "${worksheet.name}"...
`; addStatusToUI(debugLog); return worksheet.applyRangeFilterAsync(dateFilter.fieldName, { min: null, max: today }).then(() => { hasUpdatedDate = true; debugLog += `✅ הצלחה! הפילטר עודכן להיום.
`; addStatusToUI(debugLog); return true; }).catch(err => { debugLog += `❌ שגיאה בעדכון הפילטר: ${err.message}
`; addStatusToUI(debugLog); return false; }); } return false; }); }); Promise.all(promises).then(results => { const updated = results.some(r => r === true); if (!updated && !hasUpdatedDate) { debugLog += `
⚠️ סריקה הסתיימה. לא נמצא שום פילטר תאריכים מתאים באף גיליון.`; addStatusToUI(debugLog); } }); }).catch(error => { console.error("Error during Tableau init:", error); addStatusToUI("שגיאה באתחול האקסטנשיין: " + error.message); }); } function addStatusToUI(htmlMessage) { const statusDiv = document.getElementById("statusMessage"); if (statusDiv) { statusDiv.innerHTML = htmlMessage; } } })();
