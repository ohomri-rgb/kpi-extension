(function() { const VERSION = "v1.1.3"; let hasUpdatedDate = false; const checkTableauLoaded = setInterval(() => { if (typeof window.tableau !== 'undefined' && window.tableau.extensions) { clearInterval(checkTableauLoaded); initializeDateUpdater(); } }, 50); async function initializeDateUpdater() { try { await window.tableau.extensions.initializeAsync(); if (hasUpdatedDate) return; const dashboard = window.tableau.extensions.dashboardContent.dashboard; const worksheets = dashboard.worksheets; const today = new Date(); today.setHours(0, 0, 0, 0); let debugLog = `גרסה: ${VERSION}
`; debugLog += `📊 שם הדשבורד: "${dashboard.name}"
`; debugLog += `📁 מספר גיליונות שזוהו: ${worksheets.length}
`; addStatusToUI(debugLog); if (worksheets.length === 0) { debugLog += `⚠️ אזהרה: ה-API לא מזהה גיליונות עבודה פעילים בדשבורד זה.
`; addStatusToUI(debugLog); return; } // לולאה סינכרונית כדי לוודא סדר הדפסה וביצוע יציב for (const worksheet of worksheets) { debugLog += `
🔍 סורק גיליון: "${worksheet.name}"...
`; addStatusToUI(debugLog); try { const filters = await worksheet.getFiltersAsync(); debugLog += ` • נמצאו ${filters.length} פילטרים בגיליון זה.
`; for (const filter of filters) { debugLog += ` - פילטר: "${filter.fieldName}" (סוג: ${filter.filterType}, דאטה: ${filter.dataType})
`; } addStatusToUI(debugLog); // חיפוש פילטר התאריך const dateFilter = filters.find(f => f.dataType === window.tableau.DataType.Date || f.dataType === window.tableau.DataType.DateTime || f.fieldName.toLowerCase().includes('date') || f.fieldName.includes('תאריך') ); if (dateFilter) { debugLog += `⚡ מנסה לעדכן את "${dateFilter.fieldName}" בגיליון "${worksheet.name}"...
`; addStatusToUI(debugLog); await worksheet.applyRangeFilterAsync(dateFilter.fieldName, { min: null, max: today }); hasUpdatedDate = true; debugLog += `✅ הצלחה! הפילטר עודכן בהצלחה לתאריך של היום.
`; addStatusToUI(debugLog); break; // אם מצאנו ועדכנו בהצלחה, אפשר לעצור } } catch (filterError) { debugLog += `❌ שגיאה בשליפת פילטרים מגיליון זה: ${filterError.message}
`; addStatusToUI(debugLog); } } if (!hasUpdatedDate) { debugLog += `
⚠️ סריקה הסתיימה. לא נמצא שום פילטר תאריכים מתאים שיכול להתעדכן.`; addStatusToUI(debugLog); } } catch (error) { console.error("Error during Tableau init:", error); addStatusToUI("שגיאה באתחול האקסטנשיין: " + error.message); } } function addStatusToUI(htmlMessage) { const statusDiv = document.getElementById("statusMessage"); if (statusDiv) { statusDiv.innerHTML = htmlMessage; } } })();
